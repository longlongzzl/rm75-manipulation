original=NativePrimaryExecutor.__init__
def instrument(self, *args, **kwargs):
    original(self,*args,**kwargs)
    read=self._read
    count=0
    def observed():
        nonlocal count
        result=read()
        count+=1
        if count not in (28,29,48,128,228):
            return result
        import numpy as np
        from rm75_app.swm.native_bootstrap import _array
        primary=self.primary
        agent=primary.env.unwrapped.agent
        actors=dict(primary.actors)
        actors['primary_table']=primary.env.unwrapped.table_scene.table
        contacts=[]
        for name,link in primary.demo.robot.links_map.items():
            for oid, actor in actors.items():
                force=_array(agent.scene.get_pairwise_contact_forces(link,actor)).reshape(-1)
                if np.linalg.norm(force)>.0001:
                    contacts.append(dict(robot_link=name,object_id=oid,force_n=force.tolist()))
        self.emit(kind='swm_primary_contact_diagnosis',read_count=count,
            contacts=contacts,feedback=result[0],semantic_gripper_value=self._gripper_value,
            native_gripper_args={key:getattr(primary.args,key,None)
                for key in ('gripper_open','gripper_close','close_steps')},
            original_pen_grasp_predicate=bool(_array(agent.is_grasping(primary.actors['bi'],min_force=.5,max_angle=95)).reshape(-1)[0]),
            skill_holding_verified=False)
        return result
    self._read=observed

from rm75_app.swm.native_audit import CuroboNativeStageAuditor
import time,json
original_audit=CuroboNativeStageAuditor.__call__
def profiled_audit(self,plan,snapshot):
    timings={}
    restore=[]
    def wrap(owner,name,label):
        old=getattr(owner,name)
        restore.append((owner,name,old))
        def measured(*args,**kwargs):
            begin=time.monotonic()
            try:
                return old(*args,**kwargs)
            finally:
                elapsed=time.monotonic()-begin
                row=timings.setdefault(label,dict(calls=0,total_s=0.,max_s=0.))
                row['calls']+=1
                row['total_s']+=elapsed
                row['max_s']=max(row['max_s'],elapsed)
        setattr(owner,name,measured)
    for name in ('_apply','_contacts','_check_path','_verified_contact_ignores'):
        wrap(self,name,'audit.'+name)
    for name in ('_collision_diagnostics_for_states','update_scene','attach_object',
                 'update_attached_object_pose','set_measured_gripper_collision_state',
                 'set_gripper_collision_state','read_attachment_collision_state','_set_obstacle_enabled'):
        wrap(self.backend,name,'backend.'+name)
    begin=time.monotonic()
    try:
        return original_audit(self,plan,snapshot)
    finally:
        elapsed=time.monotonic()-begin
        for owner,name,old in restore:
            setattr(owner,name,old)
        (ROOT/'runtime_data/swm_release_pen_worker_07/audit_timing.json').write_text(json.dumps(dict(
            audit_elapsed_s=elapsed,timings=timings,snapshot_id=snapshot['snapshot_id'],
            source_snapshot_id=plan.source_snapshot_id,skill=plan.payload.skill),indent=2))
CuroboNativeStageAuditor.__call__=profiled_audit
from dataclasses import replace
import hashlib,numpy as np
from scipy.optimize import minimize_scalar
from rm75_app.assets import object_specs
from rm75_app.planning.gripper_collision import gripper_link_transforms
urdf=ROOT/'assets/robot_models/RM75_gripper/RM75-B/urdf/RM75-B.urdf'
def pad_depth(angle):
    transforms=gripper_link_transforms(urdf,float(angle))
    center=(transforms['left_pad'][:3,3]+transforms['right_pad'][:3,3])/2
    return float((np.linalg.inv(transforms['gripper_tcp'])@np.r_[center,1.])[2])
optimum=minimize_scalar(lambda angle:-pad_depth(angle),bounds=(0.,.91),method='bounded',options={'xatol':1e-12})
if not optimum.success:raise RuntimeError('Original pad-envelope calibration did not converge')
old_spec=object_specs.OBJECT_SPECS['bi']
if abs(old_spec.grasp_z_offset-pad_depth(.6))>1e-6:raise RuntimeError('Pen offset is not the original nominal pad calibration')
depth=max(pad_depth(0.),pad_depth(.91),pad_depth(optimum.x))
object_specs.OBJECT_SPECS['bi']=replace(old_spec,grasp_z_offset=depth)
(ROOT/'runtime_data/swm_release_pen_worker_07/pad_calibration.json').write_text(json.dumps(dict(
    domain='offline_original_URDF_kinematic_calibration_probe',original_grasp_depth_m=old_spec.grasp_z_offset,
    calibrated_grasp_depth_m=depth,maximizing_joint_angle_rad=float(optimum.x),stroke_interval_rad=[0.,.91],
    urdf_sha256=hashlib.sha256(urdf.read_bytes()).hexdigest(),physical_parameters_changed=False,
    scene_or_task_target_changed=False,hardware_connected=False),indent=2))
